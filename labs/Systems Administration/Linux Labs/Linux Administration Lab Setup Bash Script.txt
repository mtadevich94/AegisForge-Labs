#!/usr/bin/env bash
###############################################################################
# setup_lab_environment.sh
#
# Builds "The Wrkstation Chronicles" — a themed RHEL-family practical exam lab
# covering the full command scope list. Run ONCE as root on a fresh VM.
#
# Tested target: Rocky Linux 10 / AlmaLinux 10 / CentOS Stream 8 / RHEL 8
# (closest freely-downloadable RHEL-8 equivalents — Rocky 8 is the closest
#  1:1 rebuild of RHEL 8 if you don't have a Red Hat Developer subscription).
#
# Recommended VM: 2 vCPU / 2GB RAM / 20GB disk, minimal install, single NIC.
# TAKE A SNAPSHOT of the VM after this script finishes, before you start the
# lab, so you can revert and re-run modules as many times as you want.
###############################################################################
set -e

if [[ $EUID -ne 0 ]]; then
  echo "Run this as root: sudo bash setup_lab_environment.sh"
  exit 1
fi

PKG_MGR="dnf"
command -v dnf >/dev/null 2>&1 || PKG_MGR="yum"

echo "=== [1/10] Installing required packages (${PKG_MGR}) ==="
${PKG_MGR} install -y net-tools iptables-services audit vim-enhanced cronie \
  at expect openssh-server nfs-utils bind-utils policycoreutils-python-utils \
  tree man-db man-pages 2>/dev/null || true

systemctl enable --now auditd     2>/dev/null || true
systemctl enable --now crond      2>/dev/null || true
systemctl enable --now atd        2>/dev/null || true
systemctl enable --now sshd       2>/dev/null || true
systemctl enable --now nfs-server 2>/dev/null || true
systemctl enable --now iptables   2>/dev/null || true

echo "=== Building the man-db database (needed for whatis/apropos) ==="
mandb 2>/dev/null || true

# Make sure sshd will accept password auth to 127.0.0.1 (needed for the
# failed/successful-login exercises in Module 9). Back up original first.
cp -n /etc/ssh/sshd_config /etc/ssh/sshd_config.lab.bak
sed -i 's/^#\?PasswordAuthentication.*/PasswordAuthentication yes/' /etc/ssh/sshd_config
sed -i 's/^#\?PermitRootLogin.*/PermitRootLogin no/' /etc/ssh/sshd_config
systemctl restart sshd

echo "=== [2/10] Creating groups and kingdom characters (users) ==="
groupadd -f gielinor
groupadd -f arrakis

useradd -m -c "General of the God Wars Dungeon" -G gielinor General_Graardor 2>/dev/null || true
useradd -m -c "Bene Gesserit Scholar"            -G arrakis Princess_Irulan   2>/dev/null || true
useradd -m -c "Kokiri Guardian Spirit"                      Zelda_Warden      2>/dev/null || true
useradd -m -c "Fairy Companion"                             Navi              2>/dev/null || true
useradd -m -c "Bandit Camp Regular"        -s /bin/sh        Bandit            2>/dev/null || true
useradd -m -c "Mercenary"                  -s /bin/sh        Kerbe             2>/dev/null || true
useradd -m -c "Rogue Agent"                                  DAME              2>/dev/null || true
useradd -m -c "Lab Student Account"                          student           2>/dev/null || true

for u in General_Graardor Princess_Irulan Zelda_Warden Navi Bandit Kerbe DAME student; do
  echo "${u}:ChangeMe123!" | chpasswd
done

echo "=== [3/10] Building the find(1)/permissions puzzle vault ==="
mkdir -p /opt/vault/shadowfiles
for i in $(seq 1 25); do
  namelen=$(( (RANDOM % 12) + 3 ))
  name=$(tr -dc 'a-z' </dev/urandom | head -c "$namelen")
  size=$(( (RANDOM % 4000) + 1 ))
  fpath="/opt/vault/shadowfiles/${name}_${i}"
  head -c "$size" /dev/urandom > "$fpath" 2>/dev/null || true
  chmod "0$(( RANDOM % 8 ))$(( RANDOM % 8 ))$(( RANDOM % 8 ))" "$fpath" 2>/dev/null || true
done
# The actual target: exactly 10-character name, exactly 512 bytes, mode 0640
head -c 512 /dev/zero > /opt/vault/shadowfiles/Nightshade
chmod 0640 /opt/vault/shadowfiles/Nightshade
useradd -m -c "Vault Keeper (no login)" -s /sbin/nologin Vaultkeeper 2>/dev/null || true
chown Vaultkeeper:Vaultkeeper /opt/vault/shadowfiles/Nightshade

echo "=== [4/10] Building the hardlink / audit / cron scenario ==="
mkdir -p /home/General_Graardor/Documents
cat > /home/General_Graardor/Documents/warplans.txt <<'EOF'
OPERATION NIGHTSHADE — EYES ONLY
Phase 1: secure the eastern bridge.
Phase 2: rendezvous at the old mill.
EOF
chown -R General_Graardor:General_Graardor /home/General_Graardor/Documents
ln /home/General_Graardor/Documents/warplans.txt /home/student/warplans.hl
chown student:student /home/student/warplans.hl

echo "=== [5/10] Building the regex / star-chart text-processing target ==="
cat > /home/Zelda_Warden/star_charts.txt <<'EOF'
Reference coordinate set, sector scan log — DO NOT DELETE
88.4-2.9
7.100-44.2
19.3-5-2
42.07-19.3-6.88-31.5
1200.4-3.2
42.07-19.3-6.88-31.5-extra
EOF
chown Zelda_Warden:Zelda_Warden /home/Zelda_Warden/star_charts.txt
chmod 644 /home/Zelda_Warden/star_charts.txt

echo "=== [6/10] Building the passwd/shadow correlation dataset ==="
# (Bandit and Kerbe were already created with /bin/sh above for this exercise)

echo "=== [7/10] Starting a background named process for ps/lsof/kill practice ==="
pkill -f WizardTower 2>/dev/null || true
sleep 1
nohup bash -c 'exec -a WizardTower sleep 100000' >/dev/null 2>&1 &
disown

echo "=== [8/10] Setting a baseline iptables rule for the student to find/remove ==="
iptables -C OUTPUT -p tcp --sport 4444 -j DROP 2>/dev/null || \
  iptables -A OUTPUT -p tcp --sport 4444 -j DROP
service iptables save 2>/dev/null || iptables-save > /etc/sysconfig/iptables 2>/dev/null || true

echo "=== [9/10] Setting up a loopback NFS export for showmount/mount practice ==="
mkdir -p /srv/nfs_share
echo "You mounted the Wrkstation Keep archive successfully." > /srv/nfs_share/welcome.txt
grep -q "/srv/nfs_share" /etc/exports 2>/dev/null || \
  echo "/srv/nfs_share  127.0.0.1(ro,sync,no_subtree_check)" >> /etc/exports
exportfs -ra 2>/dev/null || true
systemctl restart nfs-server 2>/dev/null || true

echo "=== [10/10] Generating real failed/successful login history (lastb/last) ==="
cat > /tmp/fail_login.exp <<'EOF'
#!/usr/bin/expect -f
set user [lindex $argv 0]
set host [lindex $argv 1]
spawn ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null $user@$host
expect "password:"
send "WrongPassword!\r"
expect eof
EOF
chmod +x /tmp/fail_login.exp

declare -A FAILCOUNT=( [Bandit]=9 [Kerbe]=6 [DAME]=3 )
for u in "${!FAILCOUNT[@]}"; do
  for i in $(seq 1 "${FAILCOUNT[$u]}"); do
    /tmp/fail_login.exp "$u" "127.0.0.1" >/dev/null 2>&1 || true
  done
done

# One real successful login for Bandit, for the "validate successful logins" exercise
cat > /tmp/good_login.exp <<'EOF'
#!/usr/bin/expect -f
spawn ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null Bandit@127.0.0.1 exit
expect "password:"
send "ChangeMe123!\r"
expect eof
EOF
chmod +x /tmp/good_login.exp
/tmp/good_login.exp >/dev/null 2>&1 || true

echo "=== [Extra] Bash if/elif/else skeleton for Module 11 ==="
mkdir -p /home/student/scripts
cat > /home/student/scripts/user_status.sh <<'EOF'
#!/usr/bin/env bash
# user_status.sh <username>
# TODO (Module 11): complete the if/elif/else logic so that this script:
#   1. Prints "No such user" if the account does not exist
#   2. Else, prints "Locked"    if the account is password-locked
#   3. Else, prints "Active"   if the account exists and is not locked
#
# Hints: id <user>, passwd -S <user>, chage -l <user>

USERNAME="$1"

if id "$USERNAME" >/dev/null 2>&1; then
  :  # TODO: check lock status here
else
  :  # TODO: print "No such user"
fi
EOF
chmod +x /home/student/scripts/user_status.sh
chown -R student:student /home/student/scripts

echo ""
echo "############################################################"
echo "  Lab environment build complete."
echo "  Take a VM snapshot now, then open Lab_Guide.md."
echo "############################################################"
